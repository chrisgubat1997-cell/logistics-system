document.addEventListener("DOMContentLoaded", function () { const signupForm = document.getElementById("signupForm"); signupForm.addEventListener("submit", function (event) { event.preventDefault(); const password = document.getElementById("signupPassword").value; const confirmPassword = document.getElementById("confirmPassword").value; const message = document.getElementById("signupMessage"); if (password !== confirmPassword) { message.textContent = "Passwords do not match."; message.style.color = "red"; return; } message.textContent = "Account created successfully!"; message.style.color = "green"; setTimeout(function () { window.location.href = "login.html"; }, 1000); }); });
